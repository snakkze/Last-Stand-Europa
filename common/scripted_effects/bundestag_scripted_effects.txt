# The following scripted effect does the main job of generating the gui, by adding the seats for each party in a grid.
# This should be called every time you need to update the gui, when you add/subtract seats for example.
# Read the comments if you're interested in how it works
reload_bundestag_gui = {
	# Clears what's currently on the gui, if there's anything there.
	clear_array = bundestag_main_array

	# The array temp_seat is duplicated from the array bundestag_seat_array.
	for_each_loop = {
		array = bundestag_seat_array

		add_to_array = { temp_seat = bundestag_seat_array^i }
	}

	# It's a suprise tool for later
	set_temp_variable = { temp_index_1 = 0 }
	set_temp_variable = { temp_index_2 = 1 }

	# Set temp_size as the highest index in the array, highest index is the highest value x in temp_array^x.
	set_temp_variable = { temp_size = temp_seat^num }
	subtract_from_temp_variable = { temp_size = 1 }

	# What this does is loop through the temp_seat array, adding the current value to the next value on the list.
	# temp_seat^0 is the first value on the list, temp_seat^1 is the next and so forth.
	# the loop ends when temp_index_1 is equal to the size of the array - 1 (basically the max index).
	while_loop_effect = {
		limit = {
			check_variable = { temp_index_1 < temp_size }
		}
		# The suprise tool starts out as temp_seat^0 and temp_seat^1 and then is incremented every loop , adding the current value to the next value on the list.
		add_to_variable = { temp_seat^temp_index_2 = temp_seat^temp_index_1 }
		add_to_temp_variable = { temp_index_1 = 1 }
		add_to_temp_variable = { temp_index_2 = 1 }
	}

	# resize_array fills the bundestag_member_array with the id of the bundestag member x amount of times where x is the number of seats the member has.
	# so the array will look like this basically:
	# 1,1,1,1,1,1,1,...(repeated 90 times),2,2,2,2,2,2,2(repeated 35 times),3,3,3.. etc.
	for_loop_effect = {
		start = 0
		end = temp_seat^num

		resize_array = { array = bundestag_main_array value = bundestag_member_array^v size = temp_seat^v }
	}

	# Clear the temp_seat array used previously since it's not needed anymore.
	clear_array = temp_seat
}

# Bonus code for adding/subtracting seats
modify_bundestag_seat = {
	subtract_from_temp_variable = { temp_member = 1 }

	# check to see if result overflows
	set_temp_variable = { temp = bundestag_seat_array^temp_member }
	add_to_temp_variable = { temp = temp_seats }

	if = {
		limit = {
			check_variable = { temp > bundestag_main_array^num }
		}
		subtract_from_temp_variable = { temp = bundestag_main_array^num }
		subtract_from_temp_variable = { temp_seats = temp }
	}
	else_if = {
		limit = {
			check_variable = { temp < 0 }
		}
		multiply_temp_variable = { temp = -1 }
		subtract_from_temp_variable = { temp_seats = temp }
	}

	# adds all parties to temp_array
	for_loop_effect = {
		start = 0
		end = bundestag_seat_array^num

		add_to_temp_array = { temp_array = v }
	}

	# remove from list of parties the party to be added/subtracted
	remove_from_temp_array = { array = temp_array value = temp_member }

	# give party new seat value
	add_to_variable = { bundestag_seat_array^temp_member = temp_seats }

	# if its add
	if = {
		limit = {
			check_variable = { temp_seats > 0 }
		}
		# loop until seats to be added is 0
		while_loop_effect = {
			limit = {
				check_variable = { temp_seats > 0 }
			}

			# loops through remaining parties, subtracting 1 seat each time
			for_loop_effect = {
				start = 0
				end = temp_array^num

				if = {
					limit = {
						check_variable = { bundestag_seat_array^temp_array^v > 0 } # party must have more than 0 seats to be subtracted
						check_variable = { temp_seats > 0 } # seats to be added is bigger than 0 currently
					}
					subtract_from_variable = { bundestag_seat_array^temp_array^v = 1 } # subtract from party 1 seat
					subtract_from_temp_variable = { temp_seats = 1 } # subtract from seats to be added
				}
			}
		}
	}
	# if its subtract
	else_if = {
		limit = {
			check_variable = { temp_seats < 0 }
		}
		# loop until seats to be subtracted is 0
		while_loop_effect = {
			limit = {
				check_variable = { temp_seats < 0 }
			}

			# loops through remaining parties, adding 1 seat each time
			for_loop_effect = {
				start = 0
				end = temp_array^num

				if = {
					limit = {
						check_variable = { bundestag_seat_array^temp_array^v < bundestag_main_array^num } # party must have less than max seats to be added
						check_variable = { temp_seats < 0 } # seats to be subtracted is less than 0 currently
					}
					add_to_variable = { bundestag_seat_array^temp_array^v = 1 } # add to party 1 seat
					add_to_temp_variable = { temp_seats = 1 } # add to seats to be subtracted
				}
			}
		}
	}
	# do nothing if 0

	# Modify what shows up in the gui
	reload_bundestag_gui = yes
}

coalition_partners_calculation_2 = {
	# we can add coalition ids and by that sort, if multiple partners we add multiple times the same id to the array
	clear_array = coalition_ids
	clear_array = coalitions_array
	set_temp_variable = { current_id = 0 }

	for_each_loop = {
		array = bundestag_seat_array
		index = i
		value = v

		if = {
			limit = {
				# A Coalition member needs atleast 5 seats to become an available option
				check_variable = { v > 5 }
			}
	
			if = {
				limit = {
					check_variable = { v > 366 }
				} 
				add_to_array = { coalitions_array = i } # Every party that can rule alone
					
				add_to_array = { coalition_ids = current_id } # Each party that can rule gets its own id
				log = "[?current_id] Added once for [?i]"
				add_to_temp_variable = { current_id = 1 }
			}
		}
		

		for_each_loop = {
			array = bundestag_seat_array
			index = p
			value = z

			if = {
				limit = {
					check_variable = { v > 5 }
					check_variable = { z > 5 }
					# No coalitions with same members
					NOT = {
						check_variable = { i = p }
					}
					# Senior must have more seats then junior
					check_variable = { v > z }
				}

				set_temp_variable = { temp_seats = v }
				add_to_temp_variable = { temp_seats = z }
	
				if = {
					limit = {
						check_variable = { temp_seats > 366 }
					}
					if = {
						limit = {
							is_partner_ideology_okay = yes
						}
						add_to_array = { coalitions_array = i } # Senior of this coalition
						add_to_array = { coalitions_array = p }  # Partner of this coalition
		
						# Coalition of two parties gets two entries
						add_to_array = { coalition_ids = current_id }
						add_to_array = { coalition_ids = current_id }
						log = "[?current_id] Added twice for [?i], [?p]"
						add_to_temp_variable = { current_id = 1 }
					}
				}
			}

			for_each_loop = {
				array = bundestag_seat_array
				index = o
				value = l

				if = {
					limit = {
						check_variable = { v > 5 }
						check_variable = { z > 5 }
						check_variable = { l > 5 }
						# No coalitions with same members
						NOT = {
							check_variable = { i = p }
							check_variable = { i = o }
							check_variable = { p = o }
							check_variable = { o = p }
						}
						# Senior must have more seats then junior
						check_variable = { v > z }
						check_variable = { v > l }
						# First Junior needs more seats then second junior
						check_variable = { z > l }
					}

					set_temp_variable = { temp_seats = v }
					add_to_temp_variable = { temp_seats = z }
					add_to_temp_variable = { temp_seats = l }
	
					if = {
						limit = {
							check_variable = { temp_seats > 366 }
						}
						if = {
							limit = {
								NOT = {
									any_of = {
										array = coalitions_array
										value = vl
										index = ix

										set_temp_variable = { vl2 = vl }
										add_to_temp_variable = { vl2 = 1 }
										set_temp_variable = { vl3 = vl }
										add_to_temp_variable = { vl3 = 2 }

										set_temp_variable = { ix2 = ix }
										add_to_temp_variable = { ix2 = 1 }
										set_temp_variable = { ix3 = ix }
										add_to_temp_variable = { ix3 = 2 }
										
										# If the position of p = o and o = p then not
										check_variable = { vl = i }
										check_variable = { vl2 = o }
										check_variable = { vl3 = p }

										# Extra check that even if p and o are swapped and duplicated they must be in the id as senior i!
										check_variable = { coalition_ids^ix = coalition_ids^ix2 }
										check_variable = { coalition_ids^ix = coalition_ids^ix3 }
									}
								}
							}
							if = {
								limit = {
									is_partner_ideology_okay_2 = yes
								}
								add_to_array = { coalitions_array = i } # Senior of this coalition
								add_to_array = { coalitions_array = p }  #First Partner of this coalition
								add_to_array = { coalitions_array = o } # Second Partner of this coalition
			
								# Coalition of three parties gets three entries
								add_to_array = { coalition_ids = current_id }
								add_to_array = { coalition_ids = current_id }
								add_to_array = { coalition_ids = current_id }
								add_to_temp_variable = { current_id = 1 }
							}
						}
					}
				}
			}
		}
	}
}

debug_coalitions_print_2 = {
	clear_array = coalition_ids
	clear_array = coalitions_array
	clear_array = coalition_indexes
	coalition_partners_calculation_2 = yes

	set_temp_variable = { i = 0 }

	while_loop_effect = {
		limit = { check_variable = { i < coalition_ids^num } }
		
		set_temp_variable = { i2 = i }
		add_to_temp_variable = { i2 = 1 }
		
		set_temp_variable = { i3 = i }
		add_to_temp_variable = { i3 = 2 }

		set_temp_variable = { temp_party_id = coalitions_array^i }
		set_temp_variable = { temp_party_id_2 = coalitions_array^i2 }
		set_temp_variable = { temp_party_id_3 = coalitions_array^i3 }
		
		set_temp_variable = { temp_id = coalition_ids^i }
		set_temp_variable = { temp_id_2 = coalition_ids^i2 }
		set_temp_variable = { temp_id_3 = coalition_ids^i3 }

		# Maybe sort by seats
		add_to_array = { coalition_indexes = i }

		if = {
			limit = {
				check_variable = { temp_id = temp_id_2 } # Aktuelle ID ist die gleiche wie die nächste zeile
				NOT = {
					check_variable = { temp_id = temp_id_3 } # Aber nicht gleich wie die übernächste
				}
			}
			log = "[?i]: [?coalition_ids^i] ID (2 Party), [?temp_party_id] and [?temp_party_id_2]: [GetBundestagPartyColor1]-[GetBundestagPartyColor2] Coalition"
			add_to_temp_variable = { i = 2}
		} else_if = {
			limit = {
				check_variable = { temp_id = temp_id_3 } # Doch gleich wie übernächste
			}
		log = "[?i]: [?coalition_ids^i] ID (3 Party), [?temp_party_id], [?temp_party_id_2] and [?temp_party_id_3]: [GetBundestagPartyColor1]-[GetBundestagPartyColor2]-[GetBundestagPartyColor3] Coalition"

		} else = {
			log = "[?i]: [?coalition_ids^i] ID (1 Party), [?temp_party_id] : [GetBundestagPartyColor1] Coalition"
			add_to_temp_variable = { i = 1}
		}
	}

	if = {
		limit = {
			check_variable = { coalition_indexes^num = 0 }
		}
		add_to_array = { coalition_indexes = 999 }
	}
}
